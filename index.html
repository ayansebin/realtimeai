The three statements behind it 
-- 1. wrap the model as a tool
CREATE TOOL forecast_sku AS
  AI_FORECAST(units_sold, event_time,
    JSON_OBJECT('model' VALUE 'ttm', 'horizon' VALUE 7));

-- 2. bind a model, a prompt and tools
CREATE AGENT replenisher WITH (
  model  = 'granite-llm',
  prompt = 'Keep service level at P90. Raise a PO when projected stock falls under lead time demand. Explain in one line.',
  tools  = ['forecast_sku', 'stock_lookup', 'raise_po']
);

-- 3. run it per event, write decisions back
INSERT INTO purchase_orders
SELECT store_id, sku, AI_RUN_AGENT('replenisher', ROW(store_id, sku, units_sold, event_time))
FROM pos_sales;
