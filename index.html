The three statements behind it 
-- 1. wrap the model as a tool
CREATE TOOL forecast_sku AS
  AI_FORECAST(units_sold, event_time,
    JSON_OBJECT('model' VALUE 'ttm', 'horizon' VALUE 7));

-- 2. bind a model, a prompt and tools
CREATE AGENT replenisher WITH (
  model  = 'granite-llm',
  prompt = 'Keep service level at P90. Raise a PO when projected stock falls under lead time demand. Explain in one line.',
  tools  = ['forecast_sku', 'stock_lookup', 'raise_po']
);

-- 3. run it per event, write decisions back
INSERT INTO purchase_orders
SELECT store_id, sku, AI_RUN_AGENT('replenisher', ROW(store_id, sku, units_sold, event_time))
FROM pos_sales;

SELECT event_time, value,
  AI_FORECAST(value, event_time,
    JSON_OBJECT('model' VALUE 'ttm', 'horizon' VALUE 12))
  OVER (PARTITION BY series_id ORDER BY event_time
        RANGE BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS f
FROM your_topic;

SELECT event_time, load_kw,
  AI_FORECAST(load_kw, event_time,
    JSON_OBJECT('model' VALUE 'ttm', 'horizon' VALUE 12))
  OVER (PARTITION BY meter_id ORDER BY event_time
        RANGE BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS forecast
FROM meter_readings;
SELECT event_time, response_ms,
  AI_DETECT_ANOMALIES(response_ms, event_time,
    JSON_OBJECT('model' VALUE 'flowstate',
                'minContextSize' VALUE 20, 'maxContextSize' VALUE 512,
                'confidencePercentage' VALUE 99.0))
  OVER (PARTITION BY host_id ORDER BY event_time
        RANGE BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS anomaly
FROM server_metrics;
CREATE TABLE card_alerts AS
SELECT card_id, event_time, amount,
       anomaly.forecast_value AS expected, anomaly.lower_bound, anomaly.upper_bound, anomaly.rmse
FROM (
  SELECT card_id, event_time, amount,
    AI_DETECT_ANOMALIES(amount, event_time,
      JSON_OBJECT('model' VALUE 'ttm', 'confidencePercentage' VALUE 99.5))
    OVER (PARTITION BY card_id ORDER BY event_time
          RANGE BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS anomaly
  FROM card_payments)
WHERE anomaly.is_anomaly = TRUE;
SELECT s.sku, s.event_time AS as_of,
       step.`timestamp` AS forecast_day, step.q10 AS floor, step.q50 AS expected, step.q90 AS reorder_point
FROM (
  SELECT sku, event_time,
    AI_FORECAST(units_sold, event_time,
      JSON_OBJECT('model' VALUE 'patchtstfm', 'horizon' VALUE 7, 'maxContextSize' VALUE 730))
    OVER (PARTITION BY sku ORDER BY event_time
          RANGE BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS f
  FROM daily_sales) AS s
CROSS JOIN UNNEST(s.f.forecast) AS step (`timestamp`, mean, q10, q50, q90);
SELECT event_time, vibration_mm_s,
  AI_DETECT_ANOMALIES(vibration_mm_s, event_time,
    JSON_OBJECT('model' VALUE 'patchtstfm', 'maxContextSize' VALUE 2048,
                'upperBoundConfidencePercentage' VALUE 90.0,
                'lowerBoundConfidencePercentage' VALUE 99.9))
  OVER (PARTITION BY asset_id ORDER BY event_time
        RANGE BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS anomaly
FROM motor_telemetry;
